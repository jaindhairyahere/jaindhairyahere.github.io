Designing High-Performance Queue Data Structures in C

September 28, 2025

Introduction

Queues are fundamental data structures in systems programming, especially in scenarios involving producer-consumer patterns, inter-thread communication, and high-throughput data processing. In this comprehensive guide, we'll design and implement progressively more sophisticated queue data structures, starting from a simple single-threaded implementation and evolving to lock-free, multi-threaded variants.

Our naming convention follows industry standards:

  • S - Single
  • M - Multi
  • P - Producer
  • C - Consumer
  • T - Threaded (ST = Single Threaded, MT = Multi Threaded)

We'll implement fixed-size queues with the core operations: isFull, isEmpty, enqueue, and dequeue.

Common Interface and Data Types

Before diving into implementations, let's establish a common interface that all our queue variants will support:


#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>
#include <stdbool.h>

// Common data type for queue elements
typedef int queue_data_t;

// Queue operation results
typedef enum {
    QUEUE_SUCCESS = 0,
    QUEUE_FULL = -1,
    QUEUE_EMPTY = -2,
    QUEUE_ERROR = -3
} queue_result_t;

// Forward declaration - specific implementations will define their own structs
typedef struct queue queue_t;

// Common interface that all queue implementations will support
typedef struct {
    bool (*is_full)(const queue_t* queue);
    bool (*is_empty)(const queue_t* queue);
    queue_result_t (*enqueue)(queue_t* queue, queue_data_t data);
    queue_result_t (*dequeue)(queue_t* queue, queue_data_t* data);
    void (*destroy)(queue_t* queue);
} queue_ops_t;

// Base queue structure
struct queue {
    const queue_ops_t* ops;
    size_t capacity;
};

Implementation 1: SPSCQueue (ST) - Single Threaded

Our first implementation is a simple Single Producer Single Consumer queue for single-threaded environments. It uses two pointers to track the producer and consumer positions.

Data Structure


// Single Producer Single Consumer Queue - Single Threaded
typedef struct {
    queue_t base;           // Base queue structure
    queue_data_t* buffer;   // Circular buffer
    size_t producer_idx;    // Producer write index
    size_t consumer_idx;    // Consumer read index
} spsc_st_queue_t;

Implementation


// Forward declarations
static bool spsc_st_is_full(const queue_t* queue);
static bool spsc_st_is_empty(const queue_t* queue);
static queue_result_t spsc_st_enqueue(queue_t* queue, queue_data_t data);
static queue_result_t spsc_st_dequeue(queue_t* queue, queue_data_t* data);
static void spsc_st_destroy(queue_t* queue);

// Operation table for SPSC ST queue
static const queue_ops_t spsc_st_ops = {
    .is_full = spsc_st_is_full,
    .is_empty = spsc_st_is_empty,
    .enqueue = spsc_st_enqueue,
    .dequeue = spsc_st_dequeue,
    .destroy = spsc_st_destroy
};

// Create SPSC ST queue
queue_t* spsc_st_queue_create(size_t capacity) {
    spsc_st_queue_t* queue = malloc(sizeof(spsc_st_queue_t));
    if (!queue) return NULL;
    
    queue->buffer = malloc(sizeof(queue_data_t) * (capacity + 1)); // +1 for full detection
    if (!queue->buffer) {
        free(queue);
        return NULL;
    }
    
    queue->base.ops = &spsc_st_ops;
    queue->base.capacity = capacity;
    queue->producer_idx = 0;
    queue->consumer_idx = 0;
    
    return (queue_t*)queue;
}

static bool spsc_st_is_full(const queue_t* queue) {
    const spsc_st_queue_t* q = (const spsc_st_queue_t*)queue;
    return ((q->producer_idx + 1) % (q->base.capacity + 1)) == q->consumer_idx;
}

static bool spsc_st_is_empty(const queue_t* queue) {
    const spsc_st_queue_t* q = (const spsc_st_queue_t*)queue;
    return q->producer_idx == q->consumer_idx;
}

static queue_result_t spsc_st_enqueue(queue_t* queue, queue_data_t data) {
    spsc_st_queue_t* q = (spsc_st_queue_t*)queue;
    
    if (spsc_st_is_full(queue)) {
        return QUEUE_FULL;
    }
    
    q->buffer[q->producer_idx] = data;
    q->producer_idx = (q->producer_idx + 1) % (q->base.capacity + 1);
    
    return QUEUE_SUCCESS;
}

static queue_result_t spsc_st_dequeue(queue_t* queue, queue_data_t* data) {
    spsc_st_queue_t* q = (spsc_st_queue_t*)queue;
    
    if (spsc_st_is_empty(queue)) {
        return QUEUE_EMPTY;
    }
    
    *data = q->buffer[q->consumer_idx];
    q->consumer_idx = (q->consumer_idx + 1) % (q->base.capacity + 1);
    
    return QUEUE_SUCCESS;
}

static void spsc_st_destroy(queue_t* queue) {
    spsc_st_queue_t* q = (spsc_st_queue_t*)queue;
    free(q->buffer);
    free(q);
}

Implementation 2: SPSCQueue (MT) - Multi-Threaded with Memory Barriers

Our second implementation extends the SPSC queue for multi-threaded environments. The key insight is to use memory barriers to ensure proper ordering of memory operations, preventing race conditions without using expensive locks.

Memory Barriers in C


#ifdef _WIN32
    #include <windows.h>
    #define MEMORY_BARRIER() MemoryBarrier()
    #define COMPILER_BARRIER() _ReadWriteBarrier()
#elif defined(__GNUC__)
    #define MEMORY_BARRIER() __sync_synchronize()
    #define COMPILER_BARRIER() __asm__ __volatile__("" ::: "memory")
#else
    #error "Unsupported compiler/platform"
#endif

Data Structure


// Single Producer Single Consumer Queue - Multi Threaded
typedef struct {
    queue_t base;                    // Base queue structure
    queue_data_t* buffer;           // Circular buffer
    volatile size_t producer_idx;    // Producer write index (volatile for MT)
    volatile size_t consumer_idx;    // Consumer read index (volatile for MT)
} spsc_mt_queue_t;

Implementation


// Forward declarations
static bool spsc_mt_is_full(const queue_t* queue);
static bool spsc_mt_is_empty(const queue_t* queue);
static queue_result_t spsc_mt_enqueue(queue_t* queue, queue_data_t data);
static queue_result_t spsc_mt_dequeue(queue_t* queue, queue_data_t* data);
static void spsc_mt_destroy(queue_t* queue);

// Operation table for SPSC MT queue
static const queue_ops_t spsc_mt_ops = {
    .is_full = spsc_mt_is_full,
    .is_empty = spsc_mt_is_empty,
    .enqueue = spsc_mt_enqueue,
    .dequeue = spsc_mt_dequeue,
    .destroy = spsc_mt_destroy
};

// Create SPSC MT queue
queue_t* spsc_mt_queue_create(size_t capacity) {
    spsc_mt_queue_t* queue = malloc(sizeof(spsc_mt_queue_t));
    if (!queue) return NULL;
    
    queue->buffer = malloc(sizeof(queue_data_t) * (capacity + 1));
    if (!queue->buffer) {
        free(queue);
        return NULL;
    }
    
    queue->base.ops = &spsc_mt_ops;
    queue->base.capacity = capacity;
    queue->producer_idx = 0;
    queue->consumer_idx = 0;
    
    return (queue_t*)queue;
}

static bool spsc_mt_is_full(const queue_t* queue) {
    const spsc_mt_queue_t* q = (const spsc_mt_queue_t*)queue;
    size_t producer = q->producer_idx;
    size_t consumer = q->consumer_idx;
    return ((producer + 1) % (q->base.capacity + 1)) == consumer;
}

static bool spsc_mt_is_empty(const queue_t* queue) {
    const spsc_mt_queue_t* q = (const spsc_mt_queue_t*)queue;
    size_t producer = q->producer_idx;
    size_t consumer = q->consumer_idx;
    return producer == consumer;
}

static queue_result_t spsc_mt_enqueue(queue_t* queue, queue_data_t data) {
    spsc_mt_queue_t* q = (spsc_mt_queue_t*)queue;
    
    size_t next_producer_idx = (q->producer_idx + 1) % (q->base.capacity + 1);
    
    // Check if queue is full
    if (next_producer_idx == q->consumer_idx) {
        return QUEUE_FULL;
    }
    
    // Write data to buffer first
    q->buffer[q->producer_idx] = data;
    
    // Memory barrier ensures data is written before index update
    MEMORY_BARRIER();
    
    // Update producer index - this makes the data visible to consumer
    q->producer_idx = next_producer_idx;
    
    return QUEUE_SUCCESS;
}

static queue_result_t spsc_mt_dequeue(queue_t* queue, queue_data_t* data) {
    spsc_mt_queue_t* q = (spsc_mt_queue_t*)queue;
    
    // Check if queue is empty
    if (q->consumer_idx == q->producer_idx) {
        return QUEUE_EMPTY;
    }
    
    // Read data from buffer
    *data = q->buffer[q->consumer_idx];
    
    // Memory barrier ensures data is read before index update
    MEMORY_BARRIER();
    
    // Update consumer index
    q->consumer_idx = (q->consumer_idx + 1) % (q->base.capacity + 1);
    
    return QUEUE_SUCCESS;
}

static void spsc_mt_destroy(queue_t* queue) {
    spsc_mt_queue_t* q = (spsc_mt_queue_t*)queue;
    free(q->buffer);
    free(q);
}

Key Differences and Design Insights

Single-Threaded vs Multi-Threaded

  • Volatile Keywords: In the MT version, indices are marked volatile to prevent compiler optimizations that might cache values in registers.
  • Memory Barriers: Critical for ensuring proper ordering of memory operations. The barrier ensures that data is fully written/read before updating the index that makes it visible to other threads.
  • Lock-Free Design: Both implementations avoid locks, making them suitable for high-performance scenarios where lock contention would be a bottleneck.

Why Memory Barriers Matter

Without memory barriers, the CPU or compiler might reorder instructions for optimization. In our enqueue operation, if the index update happens before the data write, a consumer thread might read garbage data. The memory barrier prevents this reordering.

Usage Example


#include <pthread.h>
#include <unistd.h>

// Producer thread function
void* producer_thread(void* arg) {
    queue_t* queue = (queue_t*)arg;
    
    for (int i = 0; i < 100; i++) {
        while (queue->ops->enqueue(queue, i) != QUEUE_SUCCESS) {
            // Spin until space is available
            usleep(1);
        }
        printf("Produced: %d\n", i);
    }
    return NULL;
}

// Consumer thread function
void* consumer_thread(void* arg) {
    queue_t* queue = (queue_t*)arg;
    queue_data_t data;
    
    for (int i = 0; i < 100; i++) {
        while (queue->ops->dequeue(queue, &data) != QUEUE_SUCCESS) {
            // Spin until data is available
            usleep(1);
        }
        printf("Consumed: %d\n", data);
    }
    return NULL;
}

int main() {
    // Create a multi-threaded SPSC queue
    queue_t* queue = spsc_mt_queue_create(10);
    
    pthread_t producer, consumer;
    
    // Create producer and consumer threads
    pthread_create(&producer, NULL, producer_thread, queue);
    pthread_create(&consumer, NULL, consumer_thread, queue);
    
    // Wait for threads to complete
    pthread_join(producer, NULL);
    pthread_join(consumer, NULL);
    
    // Clean up
    queue->ops->destroy(queue);
    
    return 0;
}

Conclusion

We've successfully implemented two foundational queue data structures: a simple single-threaded SPSC queue and its thread-safe counterpart using memory barriers. The progression from ST to MT demonstrates the careful considerations required when designing concurrent data structures.

Key takeaways:

  • Lock-free data structures can provide excellent performance in high-throughput scenarios
  • Memory barriers are crucial for maintaining correctness in concurrent environments
  • A well-designed interface allows for easy evolution from simple to complex implementations
  • Understanding CPU and compiler optimizations is essential for low-level systems programming

These queue implementations form the foundation for more advanced concurrent data structures and are commonly used in high-performance systems, game engines, and real-time applications where predictable latency and high throughput are critical.
